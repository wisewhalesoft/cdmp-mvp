### 1.3 改後 DOM 結構

**ScoresTab 完整改寫樣本**（JSX，可直接覆蓋 L1001-1166）：

```tsx
// =========================
// Tab 2: 分數設定（唯讀總覽 — v1.3 對齊 prototype 28）
//
// 變更：
//   1. 移除列右側 pencil / trash 按鈕（編輯入口收回 Tab 2 DimensionModal）
//   2. 新增頂部唯讀說明條
//   3. 操作欄改為 external-link icon-only → 觸發 goToDimEditor(columnName)
//   4. 底部 CTA 由「新增分數區間」改為「前往 Tab 2 編輯」（次要灰底 outline）
//   5. props 移除 onAddScore / onEditScore / onDeleteScore
// =========================

function ScoresTab({
  dimensions,
  onGoToDimEditor,
}: {
  dimensions: ScoringDimUI[];
  /** 點 external-link icon 或底部 CTA 時觸發；傳 columnName 以聚焦特定維度 */
  onGoToDimEditor: (columnName?: string) => void;
}) {
  const [filterColumn, setFilterColumn] = useState<string>('ALL');

  const flatRows = useMemo(() => {
    const rows: Array<{
      columnName: string;
      matchType: MatchType | undefined;
      level1: string | null;
      level2S: string | null;
      level2E: string | null;
      score: number;
    }> = [];
    dimensions.forEach((d) => {
      if (filterColumn !== 'ALL' && d.columnName !== filterColumn) return;
      const matchType = d.matchType ?? deriveMatchType(d.scores);
      d.scores.forEach((s) => {
        rows.push({
          columnName: d.columnName,
          matchType,
          level1: s.level1,
          level2S: s.level2S,
          level2E: s.level2E,
          score: s.score,
        });
      });
    });
    return rows;
  }, [dimensions, filterColumn]);

  return (
    <div className="bg-white rounded-b-lg border border-[#E5E7EB] border-t-0 shadow-sm">
      {/* 1. 唯讀總覽說明條（prototype L351-359） */}
      <div className="px-4 py-3 border-b border-[#E5E7EB] bg-blue-50/40 flex items-start gap-2">
        <Info className="w-4 h-4 text-[#2563EB] mt-0.5 shrink-0" />
        <div className="flex-1 text-xs text-gray-700">
          <p className="font-medium text-gray-800">分數設定總覽（唯讀）</p>
          <p className="mt-0.5 text-gray-600">
            本頁面僅供瀏覽各維度分數區間。
            <strong>分數設定請於 Tab 2「計分維度」進入維度後編輯</strong>
            （新增/編輯維度 Modal 內已整合分數列編輯器）。
          </p>
        </div>
      </div>

      {/* 2. 維度欄位篩選 */}
      <div className="px-4 py-3 border-b border-[#E5E7EB] bg-gray-50/40 flex items-center gap-3">
        <label className="text-xs text-gray-500">維度欄位</label>
        <div className="relative">
          <select
            value={filterColumn}
            onChange={(e) => setFilterColumn(e.target.value)}
            aria-label="維度欄位篩選"
            className="pl-3 pr-8 py-1.5 text-sm border border-[#E5E7EB] rounded-md bg-white appearance-none focus:outline-none focus:ring-2 focus:ring-[#2563EB]/20 focus:border-[#2563EB]"
          >
            <option value="ALL">全部維度</option>
            {dimensions.map((d) => (
              <option key={d.columnName} value={d.columnName}>
                {d.columnName}
              </option>
            ))}
          </select>
          <ChevronDown className="w-4 h-4 text-gray-400 absolute right-2 top-1/2 -translate-y-1/2 pointer-events-none" />
        </div>
        <span className="ml-auto text-xs text-gray-400">
          資料表 <code>ob_levelcard_score</code>
        </span>
      </div>

      {/* 3. 表格 */}
      <div className="overflow-x-auto">
        <table className="w-full text-sm">
          <thead>
            <tr className="border-b border-[#E5E7EB] bg-gray-50/60">
              <th className="text-left px-5 py-3 font-semibold text-gray-600">column_name</th>
              <th className="text-left px-5 py-3 font-semibold text-gray-600">比對模式</th>
              <th className="text-left px-5 py-3 font-semibold text-gray-600">level1（類別型）</th>
              <th className="text-left px-5 py-3 font-semibold text-gray-600">level2_s</th>
              <th className="text-left px-5 py-3 font-semibold text-gray-600">level2_e</th>
              <th className="text-right px-5 py-3 font-semibold text-gray-600">score</th>
              <th className="text-right px-5 py-3 font-semibold text-gray-600">操作</th>
            </tr>
          </thead>
          <tbody>
            {flatRows.length === 0 && (
              <tr>
                <td colSpan={7} className="text-center py-12 text-gray-400 text-sm">
                  <Inbox className="w-8 h-8 text-gray-300 mx-auto mb-2" />
                  <p>目前無分數區間設定</p>
                  <p className="text-xs text-gray-400 mt-1">
                    請於 Tab 2「計分維度」進入維度後新增分數區間
                  </p>
                </td>
              </tr>
            )}
            {flatRows.map((r, idx) => (
              <tr
                key={idx}
                className="border-b border-[#E5E7EB] hover:bg-gray-50/50 transition"
                data-testid={`score-row-${idx}`}
              >
                <td className="px-5 py-3 font-mono text-xs text-gray-700">{r.columnName}</td>
                <td className="px-5 py-3">
                  {r.matchType ? (
                    <span
                      data-testid={`score-row-${idx}-matchtype`}
                      data-matchtype={r.matchType}
                      className={
                        'inline-flex px-2 py-0.5 text-xs rounded-full ' +
                        MATCH_TYPE_CHIP_CLASS[r.matchType]
                      }
                    >
                      {MATCH_TYPE_LABELS[r.matchType]}
                    </span>
                  ) : (
                    <span className="inline-flex px-2 py-0.5 text-xs rounded-full bg-gray-100 text-gray-500">
                      未設定
                    </span>
                  )}
                </td>
                <td className="px-5 py-3 font-mono text-sm">
                  {r.level1 ?? <span className="text-gray-300">—</span>}
                </td>
                <td className="px-5 py-3 font-mono text-sm">
                  {r.level2S ?? <span className="text-gray-300">—</span>}
                </td>
                <td className="px-5 py-3 font-mono text-sm">
                  {r.level2E ?? <span className="text-gray-300">—</span>}
                </td>
                <td
                  className={
                    'px-5 py-3 text-right font-semibold tabular-nums ' +
                    (r.score < 0 ? 'text-[#EF4444]' : '')
                  }
                >
                  {r.score}
                </td>
                <td className="px-5 py-3 text-right">
                  <button
                    type="button"
                    data-testid={`goto-dim-editor-${r.columnName}`}
                    title={`於 Tab 2 編輯維度 ${r.columnName}`}
                    aria-label={`於 Tab 2 編輯維度 ${r.columnName}`}
                    onClick={() => onGoToDimEditor(r.columnName)}
                    className="action-btn p-1.5 text-gray-500 hover:text-[#2563EB] hover:bg-blue-50 rounded transition"
                  >
                    <ExternalLink className="w-4 h-4" />
                  </button>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {/* 4. 底部說明 + 次要 CTA */}
      <div className="px-4 py-3 border-t border-[#E5E7EB] flex items-center justify-between">
        <p className="text-xs text-gray-500">
          <Info className="w-3.5 h-3.5 inline mr-1 text-gray-400" />
          各維度依其 <code>match_type</code>（CATEGORY / RANGE / COMPOSITE）顯示對應欄位。數值區間不可重疊（BR-3）
        </p>
        <button
          type="button"
          data-testid="btn-goto-dim-editor"
          onClick={() => onGoToDimEditor()}
          className="inline-flex items-center gap-1.5 px-3 py-1.5 bg-white border border-[#E5E7EB] text-gray-700 text-xs font-medium rounded-md hover:bg-gray-50 transition"
        >
          <ArrowLeft className="w-3.5 h-3.5" />
          前往 Tab 2 編輯
        </button>
      </div>
    </div>
  );
}
```

// TODO（tdd-implementation）：goToDimEditor focus 機制
// 預設方案：setTopTab('dim') + 維度 row scrollIntoView + 500ms ring 高亮
// 不引入 URL query，避免破壞 5-Tab Shell 的 context-only 設計
const goToDimEditor = useCallback((columnName?: string) => {
  setTopTab('dim');
  if (columnName) {
    // 下一個 tick 等 Tab 2 渲染完成
    requestAnimationFrame(() => {
      const row = document.querySelector(`[data-testid="dim-row-${columnName}"]`);
      if (row) {
        row.scrollIntoView({ behavior: 'smooth', block: 'center' });
        row.classList.add('ring-2', 'ring-[#2563EB]', 'ring-offset-2');
        setTimeout(() => {
          row.classList.remove('ring-2', 'ring-[#2563EB]', 'ring-offset-2');
        }, 1500);
      }
    });
    showToast({ type: 'success', message: `已切換至 Tab 2「計分維度」` });
  } else {
    showToast({ type: 'success', message: '已切換至 Tab 2「計分維度」' });
  }
}, [setTopTab, showToast]);

  cardType: string;
  cardVersion: number;
  /** 編輯模式時傳入；新增模式為 undefined */
  target?: ScoringDimUI;
  onClose: () => void;
  onSaved: () => void;
  runWriteOp: <T>(fn: () => Promise<T>) => Promise<T>;
}

interface ScoreRowDraft {
  level1: string | null;
  level2S: string | null;
  level2E: string | null;
  score: number;
}

export function DimensionModalV2({
  cardType,
  cardVersion,
  target,
  onClose,
  onSaved,
  runWriteOp,
}: DimensionModalV2Props) {
  const isEdit = !!target;
  const [columnName, setColumnName] = useState(target?.columnName ?? '');
  const [columnLabel, setColumnLabel] = useState(target?.columnLabel ?? '');
  const [matchType, setMatchType] = useState<MatchType | ''>(
    target?.matchType ?? deriveMatchType(target?.scores ?? []) ?? '',
  );
  const [rows, setRows] = useState<ScoreRowDraft[]>(
    target?.scores.map((s) => ({
      level1: s.level1,
      level2S: s.level2S,
      level2E: s.level2E,
      score: s.score,
    })) ?? [],
  );
  const [pendingMatchType, setPendingMatchType] = useState<MatchType | null>(null);
  const [apiError, setApiError] = useState<string | null>(null);

  // === match_type 切換確認（落差 6） ===
  const handleMatchTypeChange = (newMT: MatchType) => {
    if (rows.length > 0 && matchType && matchType !== newMT) {
      setPendingMatchType(newMT); // 開啟確認對話框
    } else {
      setMatchType(newMT);
    }
  };

  const confirmMatchTypeSwitch = () => {
    if (!pendingMatchType) return;
    setMatchType(pendingMatchType);
    setRows([]); // 清空
    setPendingMatchType(null);
  };

  // === 重疊偵測（落差 6，純前端 UX 提示） ===
  const overlapWarning = useMemo(() => {
    if (matchType !== 'RANGE' && matchType !== 'COMPOSITE') return null;
    return detectOverlap(rows); // 見 §6.6 pseudo-code
  }, [matchType, rows]);

  const renderThead = () => {

          <th className="text-right px-3 py-2 text-xs font-semibold text-gray-600">操作</th>
        </tr>
      );
    }
    if (matchType === 'RANGE') {
      return (
        <tr className="border-b border-[#E5E7EB] bg-gray-50/60">
          <th className="text-left px-3 py-2 text-xs font-semibold text-gray-600">level2_s（下限）</th>
          <th className="text-left px-3 py-2 text-xs font-semibold text-gray-600">level2_e（上限）</th>
          <th className="text-right px-3 py-2 text-xs font-semibold text-gray-600">score</th>
          <th className="text-right px-3 py-2 text-xs font-semibold text-gray-600">操作</th>
        </tr>
      );
    }
    if (matchType === 'COMPOSITE') {
      return (
        <tr className="border-b border-[#E5E7EB] bg-gray-50/60">
          <th className="text-left px-3 py-2 text-xs font-semibold text-gray-600">level1（分群）</th>
          <th className="text-left px-3 py-2 text-xs font-semibold text-gray-600">level2_s</th>
          <th className="text-left px-3 py-2 text-xs font-semibold text-gray-600">level2_e</th>
          <th className="text-right px-3 py-2 text-xs font-semibold text-gray-600">score</th>
          <th className="text-right px-3 py-2 text-xs font-semibold text-gray-600">操作</th>

      );

    return null;
  };

      { level1: null, level2S: null, level2E: null, score: 0 },

  };

    setRows((prev) => prev.map((r, i) => (i === idx ? { ...r, ...patch } : r)));
  };

    setRows((prev) => prev.filter((_, i) => i !== idx));
  };

    setApiError(null);
    // 驗證

      setApiError('column_name 為必填');
      return;
    }
    if (!matchType) {
      setApiError('請選擇比對模式');
      return;
    }

      await runWriteOp(() =>
        updateDimensions(cardType, cardVersion, /* payload */),
      );

    } catch (err) {

    }

        <div className="bg-white rounded-xl shadow-2xl w-full max-w-3xl relative max-h-[90vh] flex flex-col">

```ts
// TODO（tdd-implementation）：純前端重疊偵測
// 規範來源：prototype 28 L823-833（UX 提示，不鎖儲存；後端 422 SCORING_RANGE_OVERLAP 為最終仲裁）
//
// 對應 BR-3：數值區間不可重疊
//
// 規則：
//   - 僅 RANGE / COMPOSITE 模式需偵測
//   - RANGE：對所有 rows 兩兩比較 [level2_s, level2_e] 是否重疊
//   - COMPOSITE：先按 level1 分群，群內再做 RANGE 偵測
//
// 重疊定義：rowA.level2_e >= rowB.level2_s AND rowA.level2_s <= rowB.level2_e
//
// 回傳：null（無重疊）或 string（人類可讀描述）

function detectOverlap(rows: ScoreRowDraft[], matchType: MatchType): string | null {
  if (matchType !== 'RANGE' && matchType !== 'COMPOSITE') return null;

  const groups: Record<string, ScoreRowDraft[]> =
    matchType === 'RANGE'
      ? { __all__: rows }
      : rows.reduce((acc, r) => {
          const key = r.level1 ?? '__null__';
          acc[key] = acc[key] || [];
          acc[key].push(r);
          return acc;
        }, {} as Record<string, ScoreRowDraft[]>);

  for (const [groupKey, groupRows] of Object.entries(groups)) {
    const valid = groupRows.filter(
      (r) => r.level2S !== null && r.level2E !== null,
    );
    for (let i = 0; i < valid.length; i++) {
      for (let j = i + 1; j < valid.length; j++) {
        const a = valid[i];
        const b = valid[j];
        const aS = Number(a.level2S);
        const aE = Number(a.level2E);
        const bS = Number(b.level2S);
        const bE = Number(b.level2E);
        if (aE >= bS && aS <= bE) {
          const groupHint = matchType === 'COMPOSITE' ? `（分群 ${groupKey}）` : '';
          return `第 ${i + 1} 列 [${aS}, ${aE}] 與第 ${j + 1} 列 [${bS}, ${bE}] 重疊${groupHint}`;
        }
      }
    }
  }
  return null;
}

// 4 個建議 unit test case：
// 1. RANGE 無重疊：[0-100], [101-200] → 回 null
// 2. RANGE 邊界重疊：[0-100], [100-200] → 回字串（100 重疊）
// 3. RANGE 跨區間重疊：[0-100], [50-150] → 回字串
// 4. COMPOSITE 不同 level1 群內無重疊：
//    A=[0-100] + A=[101-200] + B=[0-100] → 回 null（A 群與 B 群獨立）
```
